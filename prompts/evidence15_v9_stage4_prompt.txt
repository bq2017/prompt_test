def _replace_once(old: str, new: str) -> None:
    global DIFFICULTY_RATING_PROMPT_PREFIX
    count = DIFFICULTY_RATING_PROMPT_PREFIX.count(old)
    if count != 1:
        raise ValueError(f"Stage4 prompt patch expected one match, got {count}: {old[:80]!r}")
    DIFFICULTY_RATING_PROMPT_PREFIX = DIFFICULTY_RATING_PROMPT_PREFIX.replace(old, new, 1)
